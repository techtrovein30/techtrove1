-- ============================================================================
-- QR check-in pass
-- ----------------------------------------------------------------------------
-- Gives every registered participant a personal, unforgeable QR pass that the
-- admin can scan at the venue to check them in. This is the SECOND of the two
-- check-in paths: the existing manual "Check In" / "Uncheck" buttons on
-- /wasd4381/checkin keep working untouched, and a scan flips the same
-- registration_members.attended flag.
--
-- Design notes
--   * The token is a random 32-hex-char value (128 bits) derived from
--     gen_random_uuid(), which this project already relies on. It is NOT an
--     email / reg number / registration code, so a screenshot of the pass leaks
--     nothing useful and the token cannot be guessed over the network.
--   * The token is keyed by PARTICIPANT EMAIL, not by membership row, so a
--     player who entered several events has exactly one pass and one scan
--     covers all of their memberships - mirroring the existing
--     adminTogglePlayerCheckin(email) behaviour.
--   * Scanning goes through the admin_scan_checkin SECURITY DEFINER RPC, so
--     the lookup + bulk update + audit insert are one atomic transaction and
--     the browser never needs write access to the check-in tables.
--   * Scope of a captain's team passes is resolved by JOINING registration_members
--     to registrations_internal / registrations_external on registration_id, so
--     it never depends on guessing what registration_members.user_id means.
--
-- Aligned with the live schema:
--   - registration_members has no `members` jsonb (that column lives on the two
--     registration tables), so team scope is derived from the join above.
--   - gen_random_bytes() (pgcrypto) is deliberately NOT used; the token is built
--     from gen_random_uuid(), which is core Postgres and already in use by
--     registration_members.id and notifications.id.
--   - checkin_tokens.email carries the same `email = lower(email)` CHECK the
--     schema already uses on admin_allowlist.email.
--
--   - my_checkin_token() / my_checkin_tokens() declare `email`, `display_name`
--     and `participant_type` as RETURNS TABLE columns, which PL/pgSQL treats as
--     implicit VARIABLES. The bare `email` in `on conflict (email)` is then
--     ambiguous and the RPC fails with
--       column reference "email" is ambiguous
--     so both carry `#variable_conflict use_column`. If you ever rewrite these
--     bodies, keep that line or qualify every column reference.
--
-- RUN THIS ONCE in the Supabase SQL Editor (Dashboard -> SQL Editor -> New query).
-- Safe to re-run: every statement is idempotent.
-- ============================================================================

-- === 1. Audit trail =========================================================
-- A dedicated table rather than admin_audit_log, because a scan records
-- specific counts the generic table has no columns for (how many memberships
-- were newly checked in vs. already present), and the desk needs to tell a
-- first-time arrival apart from a duplicate scan.

create table if not exists public.checkin_log (
  id               uuid primary key default gen_random_uuid(),
  email            text        not null,
  display_name     text,
  participant_type text,
  source           text        not null default 'qr'
                             check (source in ('qr', 'manual')),
  members_checked  integer     not null default 0,
  members_total    integer     not null default 0,
  already_attended integer     not null default 0,
  admin_id         uuid,
  admin_email      text,
  created_at       timestamptz not null default now()
);

create index if not exists checkin_log_email_idx
  on public.checkin_log (lower(email));
create index if not exists checkin_log_created_at_idx
  on public.checkin_log (created_at desc);

alter table public.checkin_log enable row level security;

-- Admins read the log; participants never do.
drop policy if exists "checkin_log_admin_read" on public.checkin_log;
create policy "checkin_log_admin_read" on public.checkin_log
  for select to authenticated
  using (public.is_admin());

-- === 2. One pass per participant ===========================================

create table if not exists public.checkin_tokens (
  email            text primary key check (email = lower(email)),
  -- 32 lowercase hex chars: a UUID with its dashes stripped. Generated by the
  -- database so the value is never assembled on the client.
  token            text        not null unique
                             default replace(gen_random_uuid()::text, '-', ''),
  display_name     text,
  participant_type text,
  created_at       timestamptz not null default now(),
  updated_at       timestamptz not null default now(),
  revoked_at       timestamptz
);

-- Scanning resolves the token by exact match on the indexed column.
create index if not exists checkin_tokens_token_idx
  on public.checkin_tokens (token);

-- Clients never read this table directly (all access goes through the two
-- SECURITY DEFINER functions below); it is enabled so that stays true by default.
alter table public.checkin_tokens enable row level security;

-- === 3. Stamp the check-in source on the membership rows ====================

alter table public.registration_members
  add column if not exists attended_at     timestamptz,
  add column if not exists attended_source text;

-- === 4. Keep attended_at / attended_source in sync =========================
-- A BEFORE UPDATE trigger so the manual path (adminTogglePlayerCheckin /
-- adminToggleCheckin, which only send `attended`) is stamped for free, and an
-- un-check-in clears the pair.

create or replace function public.touch_checkin_attendance()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
  if new.attended is distinct from old.attended then
    if new.attended then
      new.attended_at     = coalesce(new.attended_at, now());
      new.attended_source = coalesce(new.attended_source, 'manual');
    else
      new.attended_at     = null;
      new.attended_source = null;
    end if;
  elsif new.attended and new.attended_at is null then
    new.attended_at = now();
  end if;
  new.updated_at = now();
  return new;
end;
$$;

drop trigger if exists trg_touch_checkin_attendance on public.registration_members;
create trigger trg_touch_checkin_attendance
  before update on public.registration_members
  for each row execute function public.touch_checkin_attendance();

-- === 5. Issue a pass automatically for every registered participant ==========
-- registration_members is auto-populated by an existing trigger whenever a
-- registration is created, so hooking here means every internal and external
-- participant gets a pass with no client involvement at all.

create or replace function public.issue_checkin_token()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
  v_email text := lower(btrim(coalesce(new.email, '')));
begin
  if v_email = '' then
    return null;
  end if;

  insert into public.checkin_tokens as t (email, display_name, participant_type)
  values (v_email, btrim(coalesce(new.member_name, '')), new.participant_type)
  on conflict (email) do update
    set display_name     = coalesce(nullif(excluded.display_name, ''), t.display_name),
        participant_type = coalesce(excluded.participant_type, t.participant_type),
        updated_at        = now()
    -- Never resurrect a pass an admin deliberately revoked.
    where t.revoked_at is null;

  return null;
end;
$$;

drop trigger if exists trg_issue_checkin_token on public.registration_members;
create trigger trg_issue_checkin_token
  after insert on public.registration_members
  for each row execute function public.issue_checkin_token();

-- Backfill a pass for everyone who registered before this script was run.
insert into public.checkin_tokens (email, display_name, participant_type)
select lower(btrim(rm.email)),
       min(btrim(rm.member_name)),
       min(rm.participant_type)
  from public.registration_members rm
 where btrim(coalesce(rm.email, '')) <> ''
 group by lower(btrim(rm.email))
on conflict (email) do nothing;

-- === 6. Participant read access to their own pass ===========================
-- Routed through SECURITY DEFINER helpers rather than a self-referential RLS
-- expression, so the policy stays cheap and cannot recurse.

create or replace function public.current_user_email()
returns text
language sql
stable
security definer
set search_path = public
as $$
  select lower(t.email)
  from (
    select email from public.internal_participants where id = auth.uid()
    union all
    select email from public.external_participants where id = auth.uid()
  ) t
  where t.email is not null
  limit 1;
$$;

/**
 * The caller's own pass, created on first call so a participant who registered
 * before this script was applied still gets a QR without re-registering.
 */
create or replace function public.my_checkin_token()
returns table (
  token            text,
  email            text,
  display_name     text,
  participant_type text
)
language plpgsql
security definer
set search_path = public
as $$
#variable_conflict use_column
-- RETURNS TABLE declares every output column as an implicit PL/pgSQL VARIABLE,
-- so the bare `email` in `on conflict (email)` collides with the OUT parameter
-- named `email` and the function dies with
--   column reference "email" is ambiguous
-- before it can return a single row. use_column resolves any such tie in favour
-- of the table column, which is what every one of those references means here.
declare
  v_email text;
  v_name  text;
  v_type  text;
  v_token text;
begin
  if auth.uid() is null then
    raise exception 'Not authenticated';
  end if;

  select t.email, t.full_name, t.participant_type
    into v_email, v_name, v_type
  from (
    select email, full_name, 'internal'::text as participant_type
      from public.internal_participants where id = auth.uid()
    union all
    select email, full_name, 'external'::text as participant_type
      from public.external_participants where id = auth.uid()
  ) t
  where t.email is not null
  limit 1;

  if v_email is null then
    -- No participant profile yet: fall back to the auth email so the pass can
    -- still be rendered. It matches nothing at check-in until they register.
    v_email := lower(auth.jwt() ->> 'email');
    v_name  := coalesce(auth.jwt() -> 'user_metadata' ->> 'full_name',
                         auth.jwt() ->> 'email');
    v_type  := 'external';
  end if;

  if v_email is null then
    raise exception 'No email address on the signed-in account';
  end if;

  insert into public.checkin_tokens as t (email, display_name, participant_type)
  values (v_email, v_name, v_type)
  on conflict (email) do update
    set display_name     = coalesce(nullif(excluded.display_name, ''), t.display_name),
        participant_type = coalesce(excluded.participant_type, t.participant_type),
        updated_at        = now()
  returning t.token into v_token;

  return query select v_token, v_email, v_name, v_type;
end;
$$;

/**
 * Passes for the caller AND for everyone listed on the caller's OWN
 * registrations, so a captain can pull up and print every team member's badge
 * from their profile without ever seeing an unrelated participant's pass.
 *
 * Team scope is resolved by joining registration_members to the two
 * registration tables on registration_id and filtering on their user_id (the
 * registration owner). registration_members has no `members` jsonb column, so
 * that join is the only correct source - and it makes no assumption about what
 * registration_members.user_id holds.
 */
create or replace function public.my_checkin_tokens()
returns table (
  token            text,
  email            text,
  display_name     text,
  participant_type text,
  is_self          boolean
)
language plpgsql
security definer
set search_path = public
as $$
#variable_conflict use_column
-- See my_checkin_token() above: the OUT parameters `email` / `display_name` /
-- `participant_type` are implicit PL/pgSQL variables, and `on conflict (email)`
-- trips `column reference "email" is ambiguous` without this line.
declare
  v_uid   uuid := auth.uid();
  v_email text;
begin
  if v_uid is null then
    raise exception 'Not authenticated';
  end if;

  -- Prefer the participant row, but fall back to the signed-in JWT email.
  -- Without the fallback a participant whose internal/external row has not
  -- landed yet resolves to a NULL email, which makes is_self false for EVERY
  -- row - including their own - and the profile page then renders no QR at all.
  v_email := coalesce(
    public.current_user_email(),
    nullif(lower(btrim(auth.jwt() ->> 'email')), '')
  );

  with scope as (
    select lower(btrim(rm.email)) as email, rm.member_name, rm.participant_type
      from public.registration_members rm
      join public.registrations_internal ri
        on ri.id = rm.registration_id and ri.user_id = v_uid
    union all
    select lower(btrim(rm.email)), rm.member_name, rm.participant_type
      from public.registration_members rm
      join public.registrations_external re
        on re.id = rm.registration_id and re.user_id = v_uid
    union all
    -- The caller's own email, with a NULL name so the dedup below always
    -- prefers the real member_name from the registration rows.
    select v_email, null::text, null::text where v_email is not null
  ),
  deduped as (
    -- One row per email; prefer a real member name over the email placeholder.
    select distinct on (s.email)
           s.email,
           coalesce(nullif(s.member_name, ''), s.email) as display_name,
           s.participant_type
      from scope s
     where s.email is not null and btrim(s.email) <> ''
     order by s.email, (s.member_name is not null) desc, s.member_name
  ),
  upserted as (
    insert into public.checkin_tokens as t (email, display_name, participant_type)
    select d.email, d.display_name, d.participant_type from deduped d
    on conflict (email) do update
      set display_name = coalesce(nullif(excluded.display_name, ''), t.display_name),
          updated_at   = now()
    returning t.token, t.email, t.display_name, t.participant_type
  )
  return query
  select u.token, u.email, u.display_name, u.participant_type, (u.email = v_email)
    from upserted u
   order by (u.email = v_email) desc, u.display_name;
end;
$$;

grant execute on function public.my_checkin_token()  to authenticated;
grant execute on function public.my_checkin_tokens() to authenticated;

-- Admins keep full visibility of the pass table. There is deliberately no
-- participant policy: all participant access goes through the two functions
-- above, which is where scoping is enforced in one reviewable place.
drop policy if exists "checkin_tokens_admin_all" on public.checkin_tokens;
create policy "checkin_tokens_admin_all" on public.checkin_tokens
  for all to authenticated
  using (public.is_admin())
  with check (public.is_admin());

-- === 7. The scan itself =====================================================

/**
 * admin_scan_checkin(p_token text, p_source text default 'qr')
 *
 * Resolves a scanned/typed token to a participant and marks EVERY membership
 * row for that email as attended - mirroring the manual "Check In" button,
 * which is player-level rather than membership-level.
 *
 * Returns one row describing the outcome so the admin UI can render a success
 * card (name, events covered) or a precise failure reason:
 *   reason = 'invalid_token'  -> malformed / not a 32-hex token
 *   reason = 'not_registered' -> unknown token, or no registrations behind it
 *   reason = 'revoked'        -> pass was deactivated by an admin
 *   ok = true                 -> checked in
 *
 * `already_attended` counts rows that were already marked, so the UI can tell
 * a fresh arrival apart from a duplicate scan.
 */
create or replace function public.admin_scan_checkin(
  p_token  text,
  p_source text default 'qr'
)
returns table (
  ok               boolean,
  reason           text,
  email            text,
  display_name     text,
  participant_type text,
  members_checked  integer,
  members_total    integer,
  already_attended integer
)
language plpgsql
security definer
set search_path = public
as $$
declare
  v_token    text := lower(btrim(coalesce(p_token, '')));
  v_source   text := case when p_source = 'manual' then 'manual' else 'qr' end;
  v_email    text;
  v_name     text;
  v_type     text;
  v_revoked  timestamptz;
  v_total    integer := 0;
  v_new      integer := 0;
  v_existing integer := 0;
  v_admin    text := auth.jwt() ->> 'email';
begin
  if not public.is_admin() then
    raise exception 'Not authorised';
  end if;

  if v_token !~ '^[0-9a-f]{32}$' then
    return query select false, 'invalid_token', null, null, null, 0, 0, 0;
    return;
  end if;

  select t.email, t.display_name, t.participant_type, t.revoked_at
    into v_email, v_name, v_type, v_revoked
    from public.checkin_tokens t
   where t.token = v_token;

  if v_email is null then
    return query select false, 'not_registered', null, null, null, 0, 0, 0;
    return;
  end if;

  if v_revoked is not null then
    return query
      select false, 'revoked', v_email, v_name, v_type, 0, 0, 0;
    return;
  end if;

  select count(*), count(*) filter (where rm.attended)
    into v_total, v_existing
    from public.registration_members rm
   where lower(btrim(rm.email)) = v_email;

  if v_total = 0 then
    return query
      select false, 'not_registered', v_email, v_name, v_type, 0, 0, 0;
    return;
  end if;

  -- Mark only the rows that are not already checked in, so a second scan of the
  -- same pass reports 0 newly checked-in instead of rewriting the first scan's
  -- timestamp. (attended_at / attended_source are set explicitly because this
  -- is a SECURITY DEFINER write that deliberately bypasses the RLS path the
  -- manual buttons go through.)
  with updated as (
    update public.registration_members rm
       set attended        = true,
           attended_at     = now(),
           attended_source = v_source,
           updated_at      = now()
     where lower(btrim(rm.email)) = v_email
       and rm.attended is not true
    returning 1
  )
  select count(*) into v_new from updated;

  insert into public.checkin_log
    (email, display_name, participant_type, source,
     members_checked, members_total, already_attended, admin_id, admin_email)
  values
    (v_email, v_name, v_type, v_source,
     v_new, v_total, v_existing, auth.uid(), v_admin);

  return query
    select true, 'ok', v_email, v_name, v_type, v_new, v_total, v_existing;
end;
$$;

grant execute on function public.admin_scan_checkin(text, text) to authenticated;
