-- ============================================================================
-- TechTrove 3.0: Complete Database Security & Row Level Security (RLS) Hardening
-- ----------------------------------------------------------------------------
-- This script hardens all PostgreSQL tables and Supabase Storage against direct
-- PostgREST API tampering, privilege escalation, UTR reuse, and unauthorized
-- check-ins, WITHOUT breaking ANY existing frontend or coordinator functionality.
--
-- GUARANTEES:
-- 1. All necessary coordinator, attendance, and token tables are created if missing.
-- 2. Non-admins CANNOT promote themselves to role='admin' on profile setup/update.
-- 3. Participants CANNOT change their registered email to spoof/claim coordinator roles.
-- 4. External participants CANNOT insert into registrations_internal to bypass fees.
-- 5. External participants CANNOT insert pre-recorded ('recorded') payment status.
-- 6. Non-admins CANNOT mark themselves attended or alter certificate IDs.
-- 7. Event coordinators CAN view the registration roster for their assigned event.
-- 8. Public event catalog remains open for read, but secret QR attendance tokens
--    are protected and only readable via the secure get_event_attendance_token RPC.
-- 9. UTR uniqueness checks are strictly case-insensitive and trimmed.
-- 10. Storage uploads bucket enforces strict folder isolation (own user_id only).
-- 11. Audit logs, payment ledgers, and check-in tokens remain admin-only.
-- 12. All frontend RPCs (ensure_admin_access, username_is_taken, update_own_full_name,
--     update_own_college, check_utr_exists) are fully and safely implemented.
--
-- RUN THIS ONCE in the Supabase Dashboard -> SQL Editor -> New Query.
-- It is idempotent and completely safe to re-run.
-- ============================================================================

-- ─── 0. BASE SCHEMAS & PREREQUISITES (IF NOT EXISTS) ──────────────────────────

-- 0A. admin_allowlist
create table if not exists public.admin_allowlist (
  email text primary key check (email = lower(email)),
  created_at timestamptz not null default now()
);

alter table public.admin_allowlist enable row level security;

drop policy if exists "admin_allowlist_admin_all" on public.admin_allowlist;
create policy "admin_allowlist_admin_all"
  on public.admin_allowlist for all
  to authenticated
  using (
    exists (
      select 1
        from public.admin_allowlist a
       where lower(a.email) = lower(btrim(auth.jwt() ->> 'email'))
    )
  )
  with check (
    exists (
      select 1
        from public.admin_allowlist a
       where lower(a.email) = lower(btrim(auth.jwt() ->> 'email'))
    )
  );

-- ─── 1. CORE AUTH & SECURITY FUNCTIONS ───────────────────────────────────────

-- 1A. public.is_admin()
create or replace function public.is_admin()
returns boolean
language sql
stable
security definer
set search_path = public
as $$
  select exists (
    select 1
      from public.admin_allowlist a
     where lower(a.email) = lower(btrim(auth.jwt() ->> 'email'))
  ) or exists (
    select 1
      from (
        select email, role from public.internal_participants where id = auth.uid()
        union all
        select email, role from public.external_participants where id = auth.uid()
      ) p
     where p.role = 'admin'
  );
$$;

grant execute on function public.is_admin() to authenticated, anon;

-- 1B. public.current_user_email()
create or replace function public.current_user_email()
returns text
language sql
stable
security definer
set search_path = public
as $$
  select lower(coalesce(
    nullif(btrim(auth.jwt() ->> 'email'), ''),
    (select email from public.internal_participants where id = auth.uid()),
    (select email from public.external_participants where id = auth.uid())
  ));
$$;

grant execute on function public.current_user_email() to authenticated, anon;

-- 1C. Coordinator authorization helpers
create or replace function public.is_event_coordinator(p_event_id text)
returns boolean
language plpgsql
stable
security definer
set search_path = public
as $$
begin
  if auth.uid() is null then
    return false;
  end if;

  return exists (
    select 1
      from public.event_coordinators ec
     where ec.event_id = p_event_id
       and (
         (ec.user_id is not null and ec.user_id = auth.uid()::text)
         or lower(btrim(ec.email)) = public.current_user_email()
       )
  );
end;
$$;

grant execute on function public.is_event_coordinator(text) to authenticated;

create or replace function public.is_event_admin(p_event_id text)
returns boolean
language sql
stable
security definer
set search_path = public
as $$
  select public.is_admin() or public.is_event_coordinator(p_event_id);
$$;

grant execute on function public.is_event_admin(text) to authenticated;

-- 1D. Ensure admin access for Google OAuth allowlisted accounts
create or replace function public.ensure_admin_access()
returns boolean
language plpgsql
security definer
set search_path = public
as $$
declare
  v_email text := lower(btrim(auth.jwt() ->> 'email'));
  v_uid   uuid := auth.uid();
  v_is_allowlisted boolean;
begin
  if v_uid is null or v_email is null then
    return false;
  end if;

  select exists (
    select 1 from public.admin_allowlist a where lower(btrim(a.email)) = v_email
  ) into v_is_allowlisted;

  if not v_is_allowlisted then
    return false;
  end if;

  update public.internal_participants set role = 'admin' where id = v_uid;
  update public.external_participants set role = 'admin' where id = v_uid;

  return true;
end;
$$;

grant execute on function public.ensure_admin_access() to authenticated;

-- 1E. Check username uniqueness
create or replace function public.username_is_taken(p_username text)
returns boolean
language sql
stable
security definer
set search_path = public
as $$
  select exists (
    select 1 from public.internal_participants where lower(btrim(username)) = lower(btrim(p_username))
    union all
    select 1 from public.external_participants where lower(btrim(username)) = lower(btrim(p_username))
  );
$$;

grant execute on function public.username_is_taken(text) to anon, authenticated;

-- 1F. Secure event attendance token retriever
create or replace function public.get_event_attendance_token(p_event_id text)
returns text
language plpgsql
security definer
set search_path = public
as $$
declare
  v_token text;
begin
  if not public.is_event_admin(p_event_id) then
    raise exception 'Not authorised';
  end if;

  select attendance_token into v_token
    from public.events
   where id = p_event_id;

  if v_token is null or v_token !~ '^[0-9a-f]{32}$' then
    v_token := lower(replace(gen_random_uuid()::text, '-', ''));
    update public.events set attendance_token = v_token where id = p_event_id;
  end if;

  return v_token;
end;
$$;

grant execute on function public.get_event_attendance_token(text) to authenticated;

-- 1G. Real-time UTR Uniqueness Check RPC
create or replace function public.check_utr_exists(
  p_utr text,
  p_exclude_code text default null
)
returns boolean
language plpgsql
security definer
set search_path = public
as $$
declare
  v_clean text := btrim(lower(p_utr));
  v_exclude text := nullif(btrim(p_exclude_code), '');
begin
  if v_clean is null or length(v_clean) < 4 then
    return false;
  end if;

  return exists (
    select 1
      from public.registrations_external
     where btrim(lower(utr_number)) = v_clean
       and (v_exclude is null or registration_code <> v_exclude)
  ) or exists (
    select 1
      from public.registrations_internal
     where btrim(lower(utr_number)) = v_clean
       and (v_exclude is null or registration_code <> v_exclude)
  );
end;
$$;

grant execute on function public.check_utr_exists(text, text) to anon, authenticated, service_role;

-- ─── 8. STORAGE POLICIES ('uploads' bucket) ─────────────────────────────────
-- Strict folder isolation:
-- Users can only upload, update, and read files in their own folder:
--   uploads/payment-proofs/{auth.uid()}/*
--   uploads/id-cards/{auth.uid()}/*
-- Admins can read and delete all files in the bucket.

insert into storage.buckets (id, name, public)
values ('uploads', 'uploads', false)
on conflict (id) do update set public = false;

drop policy if exists "Authenticated users can upload to own folder" on storage.objects;
drop policy if exists "Authenticated users can update own uploads" on storage.objects;
drop policy if exists "Users and admins can read uploads" on storage.objects;
drop policy if exists "Users and admins can delete uploads" on storage.objects;

-- 8A. INSERT: User uploads to own folder
create policy "Authenticated users can upload to own folder"
on storage.objects for insert
to authenticated
with check (
  bucket_id = 'uploads' and
  (storage.foldername(name))[1] in ('payment-proofs', 'id-cards') and
  (storage.foldername(name))[2] = (auth.uid())::text
);

-- 8B. UPDATE: User updates/overwrites in own folder
create policy "Authenticated users can update own uploads"
on storage.objects for update
to authenticated
using (
  bucket_id = 'uploads' and
  (storage.foldername(name))[1] in ('payment-proofs', 'id-cards') and
  (storage.foldername(name))[2] = (auth.uid())::text
)
with check (
  bucket_id = 'uploads' and
  (storage.foldername(name))[1] in ('payment-proofs', 'id-cards') and
  (storage.foldername(name))[2] = (auth.uid())::text
);

-- 8C. SELECT: User reads own uploads, Admin reads all
create policy "Users and admins can read uploads"
on storage.objects for select
to authenticated
using (
  bucket_id = 'uploads' and (
    (
      (storage.foldername(name))[1] in ('payment-proofs', 'id-cards') and
      (storage.foldername(name))[2] = (auth.uid())::text
    )
    or public.is_admin()
  )
);

-- 8D. DELETE: User deletes own uploads, Admin deletes any
create policy "Users and admins can delete uploads"
on storage.objects for delete
to authenticated
using (
  bucket_id = 'uploads' and (
    (
      (storage.foldername(name))[1] in ('payment-proofs', 'id-cards') and
      (storage.foldername(name))[2] = (auth.uid())::text
    )
    or public.is_admin()
  )
);

-- 9A. Case-insensitive & trimmed UTR uniqueness trigger
create or replace function public.check_registration_utr_unique()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
  v_clean text := btrim(lower(coalesce(new.utr_number, '')));
begin
  if v_clean <> '' and exists (
    select 1
      from public.registrations_external
     where btrim(lower(utr_number)) = v_clean
       and registration_code <> new.registration_code
  ) then
    raise exception 'This UTR / transaction ID has already been used for another registration.';
  end if;
  return new;
end;
$$;

drop trigger if exists trg_registrations_external_utr_unique on public.registrations_external;
create trigger trg_registrations_external_utr_unique
  before insert or update of utr_number on public.registrations_external
  for each row execute function public.check_registration_utr_unique();

-- 9B. Participant update screenshot with case-insensitive duplicate UTR check
drop function if exists public.participant_update_screenshot(uuid, text);
drop function if exists public.participant_update_screenshot(uuid, text, text);
drop function if exists public.participant_update_screenshot(text, text);
drop function if exists public.participant_update_screenshot(text, text, text);

create or replace function public.participant_update_screenshot(
  p_registration_id text,
  p_screenshot_path text,
  p_utr_number text default null
)
returns void
language plpgsql
security definer
set search_path = public
as $$
declare
  v_uid   uuid := auth.uid();
  v_utr   text := btrim(lower(coalesce(p_utr_number, '')));
  v_code  text;
  v_dup   text;
begin
  if v_uid is null then
    raise exception 'Not authenticated';
  end if;

  select registration_code
    into v_code
    from public.registrations_external
   where id = p_registration_id
     and user_id = v_uid;

  if v_code is null then
    raise exception 'Registration not found';
  end if;

  if v_utr <> '' then
    select registration_code
      into v_dup
      from public.registrations_external
     where btrim(lower(utr_number)) = v_utr
       and registration_code <> v_code
     limit 1;
    if v_dup is not null then
      raise exception 'This UTR / transaction ID has already been used for another registration.';
    end if;
  end if;

  update public.registrations_external
     set payment_screenshot_path = p_screenshot_path,
         payment_screenshot_url  = p_screenshot_path,
         utr_number              = case when v_utr <> '' then p_utr_number else utr_number end,
         payment_review_note     = null,
         payment_status          = 'pending'
   where (id = p_registration_id or registration_code = v_code)
     and user_id = v_uid;

  if not found then
    raise exception 'Registration not found';
  end if;
end;
$$;

grant execute on function public.participant_update_screenshot(text, text, text) to authenticated;

-- 9C. Update Own Full Name (Profile page support)
create or replace function public.update_own_full_name(p_full_name text)
returns void
language plpgsql
security definer
set search_path = public
as $$
declare
  v_uid uuid := auth.uid();
  v_name text := btrim(coalesce(p_full_name, ''));
  v_email text := lower(btrim(auth.jwt() ->> 'email'));
begin
  if v_uid is null then
    raise exception 'Not authenticated';
  end if;

  if v_name = '' then
    raise exception 'Full name is required';
  end if;

  update public.internal_participants set full_name = v_name where id = v_uid;
  update public.external_participants set full_name = v_name where id = v_uid;

  update public.registration_members
     set member_name = v_name,
         updated_at = now()
   where user_id = v_uid
      or (v_email is not null and lower(btrim(email)) = v_email);

  update public.registrations_internal set captain_name = v_name where user_id = v_uid;
  update public.registrations_external set captain_name = v_name where user_id = v_uid;
end;
$$;

grant execute on function public.update_own_full_name(text) to authenticated;
